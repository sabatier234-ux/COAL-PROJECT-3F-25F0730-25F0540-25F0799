;Module 2: video mode and drawing primitives
; Procedures: PutPixel, FillRect
PutPixel:
        push bp
        mov bp, sp
        push ax
        push di
        push es
        cmp word [bp+8], SCREEN_W
        jae near .out
        cmp word [bp+6], SCREEN_H
        jae near .out
        mov ax, [drawseg]
        mov es, ax
        mov di, [bp+6]
        mov ax, di
        shl di, 8
        shl ax, 6
        add di, ax                  
        add di, [bp+8]             
        mov al, [bp+4]
        mov [es:di], al
.out:
        pop es
        pop di
        pop ax
        pop bp
        ret 6
FillRect:
        push bp
        mov bp, sp
        push ax
        push bx
        push cx
        push dx
        push si
        push di
        push es
        mov ax, [drawseg]
        mov es, ax
        mov bx, [bp+12]             
        mov cx, [bp+8]             
        mov si, [bp+10]            
        mov dx, [bp+6]              
        test bx, bx                 
        jns near .x1
        add cx, bx
        xor bx, bx
.x1:    mov ax, bx                 
        add ax, cx
        cmp ax, SCREEN_W
        jle near .x2
        mov cx, SCREEN_W
        sub cx, bx
.x2:    cmp cx, 0
        jle near .done
        test si, si              
        jns near .y1
        add dx, si
        xor si, si
.y1:    mov ax, si                  
        add ax, dx
        cmp ax, SCREEN_H
        jle near .y2
        mov dx, SCREEN_H
        sub dx, si
.y2:    cmp dx, 0
        jle near .done
        mov di, si
        shl di, 8
        shl si, 6
        add di, si
        add di, bx                  
        mov al, [bp+4]
        cld
.row:
        push di
        push cx
        rep stosb                   
        pop cx
        pop di
        add di, SCREEN_W            
        dec dx
        jnz near .row
.done:
        pop es
        pop di
        pop si
        pop dx
        pop cx
        pop bx
        pop ax
        pop bp
        ret 10
